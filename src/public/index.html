GUÍA DE INTEGRACIÓN ROKU BRIGHTSCRIPT (`ROKU_CLIENT_GUIDE.brs`)

Componente para buscar/descargar automáticamente y buffer local en `tmp:/`:

' ROKU BRIGHTSCRIPT OFFLINE & AUTO-SCRAPER COMPONENT

' 1. Busca la canción; si no está, el backend la descarga automáticamente.
Function SearchAndPlayTrack(searchQuery as String) as Object
    request = CreateObject("roUrlTransfer")
    request.SetUrl("https://TU-DOMINIO-RENDER.com/api/search-or-download")
    request.AddHeader("Content-Type", "application/json")
    
    jsonQuery = FormatJson({ "query": searchQuery })
    responseStr = request.PostFromString(jsonQuery)
    response = ParseJson(responseStr)
    
    if response <> invalid and response.status = "available" then
        DownloadAndPlayOffline(response.track.audio_url, response.track.id)
    else
        print "Canción en cola de descarga..."
    end if
End Function

Sub DownloadAndPlayOffline(trackUrl as String, trackId as String)
    localMp3Path = "tmp:/" + trackId + ".mp3"
    
    xfer = CreateObject("roUrlTransfer")
    xfer.SetUrl(trackUrl)
    xfer.GetToFile(localMp3Path)
    
    audioPlayer = CreateObject("roAudioPlayer")
    content = CreateObject("roAssociativeArray")
    content.Url = localMp3Path
    content.StreamFormat = "mp3"
    
    audioPlayer.AddContent(content)
    audioPlayer.Play()
End Sub

CLIENTE WEB OFFLINE INDEXEDDB (`CLIENT_OFFLINE_INDEXEDDB.js`)

class MusicOfflineClient {
  constructor() {
    this.dbName = "MusicOfflineDB";
    this.db = null;
  }

  async init() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, 1);
      request.onupgradeneeded = (e) => {
        const db = e.target.result;
        if (!db.objectStoreNames.contains("tracks")) {
          db.createObjectStore("tracks", { keyPath: "id" });
        }
      };
      request.onsuccess = (e) => {
        this.db = e.target.result;
        resolve();
      };
      request.onerror = (e) => reject(e);
    });
  }

  async fetchAndCacheTrack(trackMetadata) {
    const audioRes = await fetch(trackMetadata.audio_url);
    const audioBlob = await audioRes.blob();
    const coverRes = await fetch(trackMetadata.static_cover_url);
    const coverBlob = await coverRes.blob();

    const record = {
      id: trackMetadata.id,
      title: trackMetadata.title,
      artist: trackMetadata.artist_name,
      audioBlob, coverBlob,
      lyrics: trackMetadata.synced_lyrics_lrc,
      cachedAt: new Date()
    };

    const tx = this.db.transaction(["tracks"], "readwrite");
    tx.objectStore("tracks").put(record);
  }
}
